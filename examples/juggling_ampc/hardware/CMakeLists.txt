cmake_minimum_required(VERSION 3.18)
project(juggling_ampc_controller LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release)
set(BASE "${CMAKE_CURRENT_SOURCE_DIR}/../reference")
set(FACTOR_DIR "" CACHE PATH "Matching native factor profile")
set(POLICY_DIR "" CACHE PATH "Trained Eigen policy export")
set(EIGEN_DIR "" CACHE PATH "Eigen include root")
if(NOT EXISTS "${POLICY_DIR}/ampc_policy.hpp")
  message(FATAL_ERROR "Export a trained policy before building the hardware controller")
endif()
find_package(casadi CONFIG REQUIRED)
find_package(osqp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(juggling_controller SHARED controller_policy.cpp
 "${BASE}/cpp/src/c_api.cpp" "${BASE}/optimization/native/hil_runtime.cpp" "${BASE}/cpp/src/hil_c_api.cpp")
target_compile_definitions(juggling_controller PRIVATE JUGGLING_RUNTIME_REFERENCE=1)
target_compile_options(juggling_controller PRIVATE -O3 -DNDEBUG -march=native)
target_include_directories(juggling_controller PRIVATE "${BASE}/optimization/native" "${BASE}/cpp/include" "${FACTOR_DIR}" "${POLICY_DIR}" "${EIGEN_DIR}" "$<TARGET_PROPERTY:osqp::osqpstatic,INTERFACE_INCLUDE_DIRECTORIES>")
target_link_libraries(juggling_controller PRIVATE casadi::casadi Threads::Threads)
