cmake_minimum_required(VERSION 3.18)
project(juggling_batch_observer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(EIGEN_DIR "" CACHE PATH "Eigen include root")
find_package(OpenMP REQUIRED)
add_library(juggling_batch_observer SHARED batch.cpp)
target_compile_options(juggling_batch_observer PRIVATE -O3 -DNDEBUG)
target_include_directories(juggling_batch_observer PRIVATE ../reference/cpp/include "${EIGEN_DIR}")
target_link_libraries(juggling_batch_observer PRIVATE OpenMP::OpenMP_CXX)
