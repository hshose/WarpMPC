cmake_minimum_required(VERSION 3.18)
project(juggling_variant LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release)
set(BASE "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp")
set(EIGEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/franka_ampc/ext/eigen" CACHE PATH "Eigen")
set(FACTOR_DIR "" CACHE PATH "Exported variant functions")
option(JUGGLING_BUILD_FRANKA "Build the physical-arm harness and read-only probe" OFF)
option(JUGGLING_CONDITIONED_CONTACT_JACOBIAN "Study event-conditioned offline observer covariance" OFF)
option(JUGGLING_EXPERIMENTAL_GROUP_RHO "Offline study of pinned OSQP contact inequality penalties" OFF)
option(JUGGLING_EXPERIMENTAL_RESIDUAL_LIFT "Offline exact residual-lifted Gauss-Newton QP" OFF)
option(JUGGLING_EXPERIMENTAL_QP_DIAGNOSTICS "Offline QP graph and pinned QDLDL fill diagnostics" OFF)
find_package(casadi CONFIG REQUIRED)
find_package(osqp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_compile_options(-O3 -DNDEBUG -march=native)
add_library(juggling_controller SHARED controller.cpp "${BASE}/src/c_api.cpp" hil_runtime.cpp "${BASE}/src/hil_c_api.cpp")
target_compile_definitions(juggling_controller PRIVATE JUGGLING_RUNTIME_REFERENCE=1)
target_include_directories(juggling_controller PRIVATE . "${BASE}/include" "${FACTOR_DIR}" "${EIGEN_DIR}")
target_link_libraries(juggling_controller PRIVATE casadi::casadi osqp::osqpstatic Threads::Threads)
if(JUGGLING_EXPERIMENTAL_RESIDUAL_LIFT)
  if(JUGGLING_BUILD_FRANKA)
    message(FATAL_ERROR "Residual lifting is an offline-only numerical experiment")
  endif()
  target_sources(juggling_controller PRIVATE residual_lift.cpp)
  target_compile_definitions(juggling_controller PUBLIC JUGGLING_EXPERIMENTAL_RESIDUAL_LIFT=1)
endif()
if(JUGGLING_EXPERIMENTAL_RESIDUAL_LIFT OR JUGGLING_EXPERIMENTAL_QP_DIAGNOSTICS)
  target_sources(juggling_controller PRIVATE residual_lift_c_api.cpp)
endif()
if(JUGGLING_EXPERIMENTAL_GROUP_RHO)
  if(JUGGLING_BUILD_FRANKA)
    message(FATAL_ERROR "Experimental private group-rho bridge is restricted to offline builds")
  endif()
  include("${CMAKE_CURRENT_SOURCE_DIR}/osqp_group_rho.cmake")
  if(JUGGLING_EXPERIMENTAL_RESIDUAL_LIFT)
    target_sources(juggling_controller PRIVATE residual_lift_rho.c)
    target_compile_definitions(juggling_controller PRIVATE JUGGLING_LIFT_LINK_RHO_BRIDGE=1)
    add_executable(juggling_lift_rho_check
      "${CMAKE_CURRENT_SOURCE_DIR}/../../experiments/simplification_20261007/tuning_lift_rho_check.c"
      residual_lift_rho.c)
    get_target_property(_lift_rho_includes juggling_controller INCLUDE_DIRECTORIES)
    target_include_directories(juggling_lift_rho_check PRIVATE ${_lift_rho_includes})
    target_compile_features(juggling_lift_rho_check PRIVATE c_std_11)
    target_link_libraries(juggling_lift_rho_check PRIVATE osqp::osqpstatic m)
  endif()
endif()
if(JUGGLING_EXPERIMENTAL_QP_DIAGNOSTICS)
  if(JUGGLING_BUILD_FRANKA OR NOT JUGGLING_EXPERIMENTAL_GROUP_RHO)
    message(FATAL_ERROR "QP fill diagnostics require the verified offline group-rho build")
  endif()
  set(_lift_qdldl_dir "${JUGGLING_OSQP_PINNED_SOURCE_DIR}/algebra/_common/lin_sys/qdldl")
  set(_lift_qdldl_types "${JUGGLING_OSQP_PINNED_BUILD_DIR}/_deps/qdldl-build/include")
  juggling_verify_sha("${_lift_qdldl_dir}/qdldl_interface.h" "3406fef071315e6b11738333e8c23fa9299189f77fb0c94306d505aae6034396")
  juggling_verify_sha("${_group_rho_cache_source}/algebra/_common/lin_sys/qdldl/qdldl_interface.h" "3406fef071315e6b11738333e8c23fa9299189f77fb0c94306d505aae6034396")
  juggling_verify_sha("${_lift_qdldl_types}/qdldl_types.h" "8a22bcae13469b03ff8d8ebf557d08ca19a9e79b35d78659b9993d8fa7a1ee42")
  target_sources(juggling_controller PRIVATE residual_lift_kkt.c)
  target_include_directories(juggling_controller PRIVATE "${_lift_qdldl_dir}" "${_lift_qdldl_types}")
  target_compile_definitions(juggling_controller PRIVATE JUGGLING_LIFT_PRIVATE_DIAGNOSTICS=1)
endif()
add_library(juggling_fixture_plant SHARED "${BASE}/src/fixture_mock.cpp")
target_include_directories(juggling_fixture_plant PRIVATE . "${BASE}/include" "${FACTOR_DIR}" "${EIGEN_DIR}")
target_link_libraries(juggling_fixture_plant PRIVATE juggling_controller casadi::casadi osqp::osqpstatic Threads::Threads)
add_executable(juggling_session_check "${BASE}/src/session_check.cpp")
target_include_directories(juggling_session_check PRIVATE "${BASE}/include" "${FACTOR_DIR}")
add_executable(juggling_benchmark benchmark.cpp)
target_include_directories(juggling_benchmark PRIVATE "${BASE}/include")
target_link_libraries(juggling_benchmark PRIVATE juggling_controller)
add_executable(juggling_hil_mock "${BASE}/src/hil_mock.cpp")
target_include_directories(juggling_hil_mock PRIVATE . "${BASE}/include" "${FACTOR_DIR}" "${EIGEN_DIR}")
target_link_libraries(juggling_hil_mock PRIVATE juggling_controller casadi::casadi osqp::osqpstatic Threads::Threads)
if(JUGGLING_BUILD_FRANKA)
  find_package(Franka CONFIG REQUIRED)
  find_package(Poco REQUIRED COMPONENTS Net Foundation)
  add_executable(juggling_franka "${BASE}/src/franka_hil.cpp")
  target_include_directories(juggling_franka PRIVATE . "${BASE}/include" "${FACTOR_DIR}" "${EIGEN_DIR}")
  target_link_libraries(juggling_franka PRIVATE juggling_controller Franka::Franka Poco::Net Poco::Foundation)
  add_executable(juggling_franka_velocity "${BASE}/src/franka_hil.cpp")
  target_compile_definitions(juggling_franka_velocity PRIVATE JUGGLING_JOINT_VELOCITY=1)
  target_include_directories(juggling_franka_velocity PRIVATE . "${BASE}/include" "${FACTOR_DIR}" "${EIGEN_DIR}")
  target_link_libraries(juggling_franka_velocity PRIVATE juggling_controller Franka::Franka Poco::Net Poco::Foundation)
  add_executable(juggling_franka_probe "${BASE}/src/franka_probe.cpp")
  target_include_directories(juggling_franka_probe PRIVATE "${BASE}/include" "${FACTOR_DIR}")
  target_link_libraries(juggling_franka_probe PRIVATE Franka::Franka Poco::Net Poco::Foundation)
endif()

# Offline estimator adapter: same Eigen filter as hardware, no SDK dependency.
add_library(juggling_offline_estimator SHARED "${BASE}/src/ball_estimator_c_api.cpp")
target_include_directories(juggling_offline_estimator PRIVATE "${BASE}/include" "${EIGEN_DIR}")
if(JUGGLING_CONDITIONED_CONTACT_JACOBIAN)
  target_compile_definitions(juggling_offline_estimator PRIVATE JUGGLING_CONDITIONED_CONTACT_JACOBIAN=1)
  add_executable(juggling_observer_boundary_check "${BASE}/src/ball_estimator_boundary_check.cpp")
  target_include_directories(juggling_observer_boundary_check PRIVATE "${BASE}/include" "${EIGEN_DIR}")
  target_compile_definitions(juggling_observer_boundary_check PRIVATE JUGGLING_CONDITIONED_CONTACT_JACOBIAN=1)
  add_executable(juggling_observer_boundary_legacy "${BASE}/src/ball_estimator_boundary_check.cpp")
  target_include_directories(juggling_observer_boundary_legacy PRIVATE "${BASE}/include" "${EIGEN_DIR}")
endif()
