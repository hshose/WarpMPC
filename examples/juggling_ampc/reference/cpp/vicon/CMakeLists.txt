cmake_minimum_required(VERSION 3.18)
project(juggling_vicon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(VICON_SDK_DIR "" CACHE PATH "Vicon DataStream SDK source from libmotioncapture/deps")
add_compile_options(-O3 -DNDEBUG -march=native)
add_subdirectory("${VICON_SDK_DIR}" sdk EXCLUDE_FROM_ALL)
add_executable(juggling_vicon ../src/vicon_stream.cpp)
target_compile_definitions(juggling_vicon PRIVATE _BUILD_STATIC_LIB)
target_include_directories(juggling_vicon PRIVATE ../include "${VICON_SDK_DIR}/Vicon/CrossMarket/DataStream")
target_link_libraries(juggling_vicon PRIVATE ViconDataStreamSDK_CPP)
