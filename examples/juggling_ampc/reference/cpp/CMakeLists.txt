cmake_minimum_required(VERSION 3.18)
project(franka_juggling_cpp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
option(JUGGLING_NATIVE "Optimize for the build machine CPU" ON)
option(JUGGLING_BUILD_FACTORS "Compile the exported CasADi stage functions" ON)
option(JUGGLING_BUILD_FRANKA "Build the optional physical-arm harness" OFF)
set(FACTOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/cpp_factors" CACHE PATH "Generated factor directory")
set(EIGEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../resources/franka_ampc/ext/eigen" CACHE PATH "Eigen headers")
find_package(casadi CONFIG REQUIRED)
find_package(osqp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_compile_options(-O3 -DNDEBUG)
if(JUGGLING_NATIVE)
  add_compile_options(-march=native)
endif()
if(JUGGLING_BUILD_FACTORS)
  file(GLOB FACTOR_SOURCES CONFIGURE_DEPENDS "${FACTOR_DIR}/*.c")
  add_library(juggling_factors SHARED ${FACTOR_SOURCES})
  target_link_libraries(juggling_factors PRIVATE m)
  set_target_properties(juggling_factors PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${FACTOR_DIR}")
endif()
add_library(juggling_controller SHARED src/controller.cpp src/c_api.cpp src/hil_runtime.cpp src/hil_c_api.cpp)
target_include_directories(juggling_controller PUBLIC include PRIVATE "${FACTOR_DIR}" "${EIGEN_DIR}")
target_link_libraries(juggling_controller PRIVATE casadi::casadi osqp::osqpstatic Threads::Threads)
# Public C interface keeps the controller callable from Python or a libfranka
# executable without exposing CasADi/Eigen objects across that boundary.
set_target_properties(juggling_controller PROPERTIES BUILD_RPATH_USE_ORIGIN ON)
add_executable(juggling_benchmark src/benchmark.cpp)
target_include_directories(juggling_benchmark PRIVATE include)
target_link_libraries(juggling_benchmark PRIVATE juggling_controller)
add_executable(juggling_hil_mock src/hil_mock.cpp)
target_include_directories(juggling_hil_mock PRIVATE include "${FACTOR_DIR}" "${EIGEN_DIR}")
target_link_libraries(juggling_hil_mock PRIVATE juggling_controller casadi::casadi osqp::osqpstatic Threads::Threads)
if(JUGGLING_BUILD_FRANKA)
  find_package(Franka CONFIG REQUIRED)
  find_package(Poco REQUIRED COMPONENTS Net Foundation)
  add_executable(juggling_franka src/franka_hil.cpp)
  target_include_directories(juggling_franka PRIVATE include "${FACTOR_DIR}" "${EIGEN_DIR}")
  target_link_libraries(juggling_franka PRIVATE juggling_controller Franka::Franka Poco::Net Poco::Foundation)
  add_executable(juggling_franka_probe src/franka_probe.cpp)
  target_include_directories(juggling_franka_probe PRIVATE include "${FACTOR_DIR}")
  target_link_libraries(juggling_franka_probe PRIVATE Franka::Franka Poco::Net Poco::Foundation)
endif()
