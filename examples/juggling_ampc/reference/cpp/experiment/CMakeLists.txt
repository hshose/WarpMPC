cmake_minimum_required(VERSION 3.18)
project(juggling_ball_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(FACTOR_DIR "" CACHE PATH "Compiled MPC factors")
set(CONTROLLER_DIR "" CACHE PATH "Compiled controller library")
set(EIGEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/franka_ampc/ext/eigen" CACHE PATH "Eigen headers")
set(BOOST_INCLUDE_DIR "" CACHE PATH "Boost headers (property_tree, header only)")
find_package(casadi CONFIG REQUIRED)
find_package(osqp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_compile_options(-O3 -DNDEBUG -march=native)
file(SHA256 "${FACTOR_DIR}/model_config.hpp" MODEL_SHA256)
add_executable(juggling_ball_cpp ../src/ball_experiment.cpp)
target_compile_definitions(juggling_ball_cpp PRIVATE JUGGLING_MODEL_SHA256="${MODEL_SHA256}")
target_include_directories(juggling_ball_cpp PRIVATE ../include ../../optimization/native "${FACTOR_DIR}" "${EIGEN_DIR}" "${BOOST_INCLUDE_DIR}")
target_link_libraries(juggling_ball_cpp PRIVATE "${CONTROLLER_DIR}/libjuggling_controller.so" casadi::casadi osqp::osqpstatic Threads::Threads)
add_executable(juggling_ball_estimator_check ../src/ball_estimator_check.cpp)
target_include_directories(juggling_ball_estimator_check PRIVATE ../include "${EIGEN_DIR}")
# Device doubles exercise full supervisor lifecycle without SDK/network access.
add_executable(juggling_mock_experiment_devices ../src/mock_experiment_devices.cpp)
target_compile_definitions(juggling_mock_experiment_devices PRIVATE JUGGLING_MODEL_SHA256="${MODEL_SHA256}")
target_include_directories(juggling_mock_experiment_devices PRIVATE ../include ../../optimization/native "${FACTOR_DIR}" "${EIGEN_DIR}" "${BOOST_INCLUDE_DIR}")
target_link_libraries(juggling_mock_experiment_devices PRIVATE "${CONTROLLER_DIR}/libjuggling_controller.so" casadi::casadi osqp::osqpstatic Threads::Threads)
