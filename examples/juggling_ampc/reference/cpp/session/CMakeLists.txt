cmake_minimum_required(VERSION 3.18)
project(juggling_session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(FACTOR_DIR "" CACHE PATH "Optimized variant containing model_config.hpp")
add_compile_options(-O3 -DNDEBUG -march=native)
find_package(Franka REQUIRED)
find_package(Poco REQUIRED COMPONENTS Net Foundation)
find_package(Threads REQUIRED)
add_executable(juggling_franka_session ../src/franka_session.cpp)
file(SHA256 "${FACTOR_DIR}/model_config.hpp" MODEL_SHA256)
target_compile_definitions(juggling_franka_session PRIVATE JUGGLING_MODEL_SHA256="${MODEL_SHA256}")
target_include_directories(juggling_franka_session PRIVATE ../include "${FACTOR_DIR}")
target_link_libraries(juggling_franka_session PRIVATE Franka::Franka Poco::Net Poco::Foundation Threads::Threads)
# Hardware-free cross-process transport/overflow/protocol checks.
add_executable(juggling_native_transport_check ../src/native_transport_check.cpp)
target_include_directories(juggling_native_transport_check PRIVATE ../include)
target_link_libraries(juggling_native_transport_check PRIVATE Threads::Threads)
